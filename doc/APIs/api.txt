exploitation_kimind.pl
---------------------------------------
• recup_infos(
				Joueur,		# (input, string) identifiant du joueur
				Dossier,		# (input, string) dossier dans lequel se trouvent les informations sur le joueur
				Infos		# (output, atom) les infos sur le joueur sous la forme infos(TempsMoyenGlobal, TempsMoyenJoueur, NbCoupsMoyenGlobal, NbCoupsMoyenJoueur)
			 )
	Description : Ce prédicat renvoie le temps moyen du joueur pour une partie Kimind, 
		      	  son nombre de coups moyen ainsi que le temps moyen global (de tous les
		          joueurs) et le nombre de coups moyen global.
	Appel       : Dans SWI-Prolog :
				  recup_infos('J1', '../stats_kimind', Stats).
	Résultat    : Stats = infos(135.47,51.33,2.46,2.16).

bk_partie.pl
---------------------------------------
• instance_characteristics(
							Instance,			# (input, atom) énoncé de la partie sous la forme instance(Sum, Order)
							Solution,			# (input, list) solution de la partie
							Characteristics 	# (output, atom) caractéristiques de la partie sous la forme characteristics(Difficulty, Domain_characteristic, Colours_order)
			  			  )
	Description : La méthode permet de récupérer les caractéristiques objectives d'une partie à
				  l'aide de l'énoncé (somme + ordre) et de sa solution.
	Appel       : Dans SWI-Prolog : 
				  instance_characteristics(
											instance(22,'r<b<g<y'),
											[c(0, red, 2), c(1, blue, 4), c(2, green, 7), c(3, yellow,9)],
											Characteristics
				    					  ).
	Résultat    : Characteristics = characteristics(very_difficult, dichotomic, same).

bk_joueur_name.pl	(valable pour tous les joueurs en remplaçant name par leur prénom)
---------------------------------------
• follows_colours_order(Name).				# indique que le joueur pose les couleurs dans le même ordre que l'énoncé 
• follows_reversed_colours_order(Name). 		# indique que le joueur pose les couleurs dans l'ordre inverse de l'énoncé
• respects_colours_constraint(Name).			# indique que le joueur respecte la contrainte sur l'ordre des couleurs
• respects_sum_constraint(Name).				# indique que le joueur respecte la contrainte de la somme
• possible_strategy(Name).					# indique que le joueur joue des coups possibles
• dichotomic_player(Name).					# indique que le joueur a une stratégie dichotomique
• limit_values_player(Name).					# indique que le joueur utilise des valeurs limites par rapport aux domaines
											# Ces faits sont valables si et seulement si le joueur joue comme indiqué dans au moins 75% des parties jouées.
• fast_player(Name).							# indique que le joueur a une durée moyenne de partie plus courte que la durée moyenne globale (de tous les joueurs)
• strong_player(Name).						# indique que le joueur a un nombre de coups moyen dans une partie plus faible que la moyenne globale (de tous les joueurs)
• slow_player(Name).							# indique que le joueur a une durée moyenne de partie plus longue que la durée moyenne globale (de tous les joueurs)
• weak_player(Name).							# indique que le joueur a un nombre de coups moyen dans une partie plus élevé que la moyenne globale (de tous les joueurs)

bk_enonce.pl
---------------------------------------
• unique_solution_instance(
							Instance	# (input, atom) énoncé de la partie sous la forme instance(Sum, Order)
			  			  )
	Description : Le prédicat indique si oui ou non un énoncé n'a qu'une seule solution.
	Appel       : Dans SWI-Prolog :
				  unique_solution_instance(instance(22,'r<b<g<y')).
	Résultat    : False.
• difficulty_instance(
						Instance,		# (input, atom) énoncé de la partie sous la forme instance(Sum, Order)
						Nb_categories,	# (input, int) nombre de catégories de difficultés souhaité (3 ou 5)
						Difficulty		# (output, atom) catégorie de difficulté de l'énoncé
					 )
	Description : Le prédicat indique la difficulté d'un énoncé selon le nombre de catégories souhaité.
	Appel       : Dans SWI-Prolog :
				  difficulty_instance(
										instance(22,'r<b<g<y'),
										3,
										Difficulty
									 ).
	Résultat    : Difficulty = very_difficult.

• dichotomic_domains_solution(
								Instance,	# (input, atom) énoncé de la partie sous la forme instance(Sum, Order)
								Solution	# (input, list) solution de la partie
			     			 )
	Description : La méthode permet de savoir si la solution de l'énoncé se prête à une
		      	  stratégie dichotomique ou non.
	Appel       : Dans SWI-Prolog :
				  dichotomic_domains_solution(
												instance(22,'r<b<g<y'),
												[c(0, red, 2), c(1, blue, 4), c(2, green, 7), c(3, yellow,9)]
											 ).
	Résultat    : True.
